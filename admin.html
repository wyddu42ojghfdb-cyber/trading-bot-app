// 🌐 الربط المباشر الموثوق بسيرفر الـ Render الحي الخاص بك لتبادل البيانات const SERVER_URL = "https://onrender.com"; // تحديث دالة جلب الطلبات لقراءة وتمرير البيانات المباشرة دون حظر async function talepleriGetir() { try { // إضافة مسار التوثيق لضمان جلب مصفوفة المعاملات مباشرة const response = await fetch(`${SERVER_URL}/api/yonetici/talepler`, { method: "POST", headers: { "Content-Type": "application/json" } }); if (response.ok) { const data = await response.json(); // تحديث قائمة بطاقة الإيداع الخضراء حياً من السيرفر const ideacAlani = document.getElementById('ideac-listesi'); if (data.yatirmalar && data.yatirmalar.length > 0) { let htmlContent = ""; data.yatirmalar.forEach(talep => { htmlContent += `
👤 اسم المستخدم: ${talep.kullanici}
💰 المبلغ المطلوب: $${talep.miktar} USDT
`; }); ideacAlani.innerHTML = htmlContent; } else { ideacAlani.innerHTML = "

لا توجد طلبات إيداع معلقة حالياً من المستخدمين.

"; } // تحديث قائمة بطاقة السحب البرتقالية حياً من السيرفر const cekmeAlani = document.getElementById('cekme-listesi'); if (data.cekmeler && data.cekmeler.length > 0) { let htmlContent = ""; data.cekmeler.forEach(talep => { htmlContent += `
👤 اسم المستخدم: ${talep.kullanici}
💸 المبلغ المراد سحبه: $${talep.miktar} USDT
🔑 عنوان محفظة المستلم: ${talep.cuzdan}
`; }); cekmeAlani.innerHTML = htmlContent; } else { cekmeAlani.innerHTML = "

لا توجد طلبات سحب معلقة حالياً من المستخدمين.

"; } } } catch (error) { console.error("Veri çekme hatası:", error); } }
